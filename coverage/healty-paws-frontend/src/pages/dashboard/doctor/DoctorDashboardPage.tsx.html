                  </div>
                ))}
              </div>
            </DashboardSection>
          )}
        </div>
 
        {isAvailabilityModalOpen && (
          <AvailabilityModal
            initialAvailability={doctorAvailability}
            onSave={handleSaveAvailability}
            onClose={() => setIsAvailabilityModalOpen(false)}
          />
        )}
      </div>
    </ProtectedRoute>
  );
}
